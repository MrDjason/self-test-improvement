```vhdl
-- entity
ENTITY e_name is -- e_name 实体名称
    PORT(
        in1, in2, in3: IN STD_LOGIC;
        out1, out2 : OUT STD_LOGIC
    );
END ENTITY e_name;
```

```vhdl
-- architecture

    --
    --
END ARCHITECTURE arch_name;
```

- 常用并行描述语句伪代码总结

```markdown
- 信号带入语句
- 条件信号带入语句（条件信号赋值语句）
目标信号量 <= 表达式1 when 条件1 else
             表达式2 when 条件2 else
             表达式3 when 条件3 else
             表达式;

- 选择信号带入语句（选择信号赋值语句）
with 选择信号 select
目的信号量 <= 表达式1 when 值1,
             表达式2 when 值2,
             表达式3 when 值3,
             表达式n when others;
```

### 条件信号带入语句——半加器

```vhdl
-- half plus 
library ieee;

use ieee.std_logic_1164.all;

entity halfplus_vhd is
port(
	ain,bin : in std_logic;
	sum,co : out std_logic -- out 端口方向
);
end halfplus_vhd;

architecture rtl of halfplus_vhd is
begin
--	sum <= ain xor bin;
--	co <= ain and bin;
	sum <= '0' when ain='0' and bin = '0' else
		   '1' when ain='1' and bin = '0' else
		   '1' when ain='0' and bin = '1' else
		   '0';
	co <= '0' when ain = '0' and bin = '0' else
		  '0' when ain = '1' and bin = '0' else
		  '0' when ain = '0' and bin = '1' else
		  '1';

end rtl;
```


### 选择信号带入语句——半加器

```vhdl
-- half plus
library ieee;

use ieee.std_logic_1164.all;

entity halfplus_vhd is
port(
	ain,bin : in std_logic;
	sum,co : out std_logic -- out 端口方向
);
end halfplus_vhd;

architecture rtl of halfplus_vhd is
signal sin : std_logic_vector(1 downto 0); -- 矢量信号 有宽度 两位二进制矢量信号
begin
--	sum <= ain xor bin;
--	co <= ain and bin;
	sin <= ain & bin;
	with sin select
	sum <= '0' when "00",
		   '1' when "01",
		   '1' when "10",
		   '0' when "11",
		   'Z' when others;
	with sin select
	co <= '0' when "00",
		  '0' when "01",
		  '0' when "10",
		  '1' when "11",
		  'Z' when others;
end rtl;
```

- 选择信号带入语句
```vhdl
-- decoder 24

library ieee;
use ieee.std_logic_1164.all;

entity decoder24 is
port(
	a0,a1 : in std_logic;
	y0,y1,y2,y3 : out std_logic
);
end decoder24;

architecture rtl of decoder24 is
signal yout : std_logic_vector(3 downto 0);
signal ain : std_logic_vector(1 downto 0);
begin
	ain <= a0 & a1;
	y0 <= yout(0); y1<=yout(1); y2 <= yout(2); y3 <= yout(3);
	
	yout <= "1110" when ain="00" else
			"1101" when ain="01" else
			"1011" when ain="10" else
			"0111" when ain="11" else
			"ZZZZ";
end rtl;
```

```vhdl
signal sin std_logic_vector(1 downto 0);
-- std_logic：单个值，可以取'0'、'1'、'Z'、'X'
-- std_logic_vector(3 downto 0)：由4个std_logic位组成的向量
-- (3 downto 0)：表示最高位索引3，最低位索引0
-- yout(3)、yout(2)、yout(1)、yout(0)表示4个独立比特位
-- yout <= "1010" 表示yout(3)=1、yout(2)=0、yout(1)=1、yout(0)=0
```

### <= 的两种用途

(1) 信号赋值

用于将右边的值赋给左边的信号。

```vhdl
signal a, b : std_logic;
a <= b;               -- 将 b 的值赋给 a
cnt_reg <= cnt_reg + 1;   -- 信号累加赋值
```

(2) 关系运算符（小于等于）

用于条件判断，返回布尔值（true 或 false）。

```vhdl
if count <= 10 then   -- 如果 count 小于或等于 10
    -- 执行某些操作
end if;
```

### => 的唯一用途：关联符号

`=>` 用于将**形式名称**与**实际对象**相关联，主要出现在以下两种场景：

(1) 聚合赋值（Aggregate）
- 给向量或记录的各个元素指定值。
- 左边是元素名或位置，右边是要赋的值。

**示例**：
```vhdl
signal data : std_logic_vector(3 downto 0);
data <= (0 => '1', 1 => '0', 2 => '1', 3 => '0');
-- 也可以使用 others
data <= (others => '0');   -- 所有位赋 '0'
```

(2) 端口映射或泛型映射
- 在实例化元件或模块时，将顶层端口/泛型与子模块端口/泛型关联。

**示例**：
```vhdl
counter_inst : entity work.counter
    generic map (WIDTH => 8)          -- 将泛型 WIDTH 映射为 8
    port map (
        clk   => clk_signal,           -- 将端口 clk 连接到 clk_signal
        rst   => rst_signal,
        en    => en_signal,
        count => count_signal
    );
```

```vhdl
library ieee;
use ieee.std_logic_1164.all;

entity test3 is
port(
	a0, a1, a2, a3 : in std_logic;
	y0, y1, y2, y3, y4, y5, y6 : out std_logic
);
end test3;

architecture rtl of test3 is
signal yout : std_logic_vector(6 downto 0);
signal ain : std_logic_vector(3 downto 0);
begin
	ain <= a3 & a2 & a1 & a0;
	y0 <= yout(0); y1 <= yout(1); y2 <= yout(2); y3 <= yout(3); y4 <= yout(4); y5 <= yout(5); y6 <= yout(6);
	yout <= "0001110" when ain="1111" else
			"0000110" when ain="1110" else
			"0100001" when ain="1101" else
			"1000110" when ain="1100" else
			"0000011" when ain="1011" else
			"0001000" when ain="1010" else
			"0010000" when ain="1001" else
			"0000000" when ain="1000" else
			"1111000" when ain="0111" else
			"0000010" when ain="0110" else
			"0010010" when ain="0101" else
			"0011001" when ain="0100" else
			"0110000" when ain="0011" else
			"0100100" when ain="0010" else
			"1111001" when ain="0001" else
			"1000000" when ain="0000" else
			"0000000";
end rtl;
```
